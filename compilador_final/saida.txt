static char* copiarLexema(const char *s) {
    size_t len = strlen(s) + 1;
    char *copia = (char*) malloc(len);
    if (copia != NULL) {
        memcpy(copia, s, len);
    }
    return copia;
}

int adaptaToken(void) {
    TokenType token = getToken();

    switch (token) {
        case INT:             return T_INT;
        case VOID:            return T_VOID;
        case ELSE:            return T_ELSE;
        case IF:         yylval.num = lineno; return T_IF;
        case WHILE:      yylval.num = lineno; return T_WHILE;
        case RETURN:     yylval.num = lineno; return T_RETURN;
        case ATRIBUICAO: yylval.num = lineno; return T_ATRIBUICAO;
        
        case IDENTIFICADOR:
            yylval.id.texto = copiarLexema(tokenString);
            yylval.id.linha = lineno;
            return T_IDENTIFICADOR;

        case NUMERO:
            yylval.num = atoi(tokenString);
            return T_NUMERO;

        case IGUAL:           return T_IGUAL;
        case MENOR:           return T_MENOR;
        case MENOR_IGUAL:     return T_MENOR_IGUAL;
        case MAIOR:           return T_MAIOR;
        case MAIOR_IGUAL:     return T_MAIOR_IGUAL;
        case DIFERENTE:       return T_DIFERENTE;
        case MAIS:            return T_MAIS;
        case MENOS:           return T_MENOS;
        case VEZES:           return T_VEZES;
        case DIVISAO:         return T_DIVISAO;
        case PONTO_VIRGULA:   return T_PONTO_VIRGULA;
        case VIRGULA:         return T_VIRGULA;
        case ABRE_PARENTESE:  return T_ABRE_PARENTESE;
        case FECHA_PARENTESE: return T_FECHA_PARENTESE;
        case ABRE_COLCHETE:   return T_ABRE_COLCHETE;
        case FECHA_COLCHETE:  return T_FECHA_COLCHETE;
        case ABRE_CHAVE:      return T_ABRE_CHAVE;
        case FECHA_CHAVE:     return T_FECHA_CHAVE;
        case ENDFILE:         return 0;

        case ERROR:
        default:
            reportarErro(FASE_LEXICA, lineno, "Token ou caractere invalido", tokenString);
            return T_ERROR;
    }
}

int yylex(void) {

static NoAST* alocarNo(TipoNoAST kind, int linha) {
    NoAST *no = (NoAST*) calloc(1, sizeof(NoAST));
    if (!no) {
        fprintf(stderr, "Erro fatal: sem memoria para AST.\n");
        exit(1);
    }
    no->kind = kind;
    no->linha = linha;
    no->tipo = TipoVoid;
    return no;
}

NoAST* anexarIrmao(NoAST *lista, NoAST *novo) {
    if (lista == NULL) return novo;
    NoAST *atual = lista;
    while (atual->irmao != NULL) atual = atual->irmao;
    atual->irmao = novo;
    return lista;
}

void imprimirASTDecorada(NoAST *raiz, int nivel) {
    for (NoAST *atual = raiz; atual != NULL; atual = atual->irmao) {
        for (int i = 0; i < nivel; i++) printf("  ");

        switch (atual->kind) {
            case NO_DECL_VAR:
                printf("DeclVar(%s %s%s) [tipo=%s]\n",
                       nomeTipo(atual->tipoDecl), atual->nome,
                       atual->ehArray ? "[]" : "", nomeTipo(atual->tipo));
                break;
            case NO_DECL_FUNC:
                printf("DeclFunc(%s %s) [tipo=%s]\n",
                       nomeTipo(atual->tipoDecl), atual->nome, nomeTipo(atual->tipo));
                break;
            case NO_PARAM:
                printf("Param(%s %s%s) [tipo=%s]\n",
                       nomeTipo(atual->tipoDecl), atual->nome,
                       atual->ehArray ? "[]" : "", nomeTipo(atual->tipo));
                break;
            case NO_COMPOSTO:   printf("BlocoComposto\n"); break;
            case NO_IF:         printf("If\n"); break;
            case NO_WHILE:      printf("While\n"); break;
            case NO_RETURN:     printf("Return [tipo=%s]\n", nomeTipo(atual->tipo)); break;
            case NO_ATRIBUICAO: printf("Atribuicao (=) [tipo=%s]\n", nomeTipo(atual->tipo)); break;
            case NO_OP_BINARIO: printf("OpBinario(%s) [tipo=%s]\n", nomeOperador(atual->op), nomeTipo(atual->tipo)); break;
            case NO_VAR:        printf("Var(%s%s) [tipo=%s]\n", atual->nome, atual->filho[0] ? "[...]" : "", nomeTipo(atual->tipo)); break;
            case NO_LITERAL:    printf("Literal(%d) [tipo=%s]\n", atual->valor, nomeTipo(atual->tipo)); break;
            case NO_CHAMADA:    printf("ChamadaFunc(%s) [tipo=%s]\n", atual->nome, nomeTipo(atual->tipo)); break;
        }

        for (int i = 0; i < MAX_FILHOS; i++) {
            if (atual->filho[i] != NULL) {
                imprimirASTDecorada(atual->filho[i], nivel + 1);
            }
        }
    }
}

void liberarAST(NoAST *raiz) {
    if (raiz == NULL) return;
    for (int i = 0; i < MAX_FILHOS; i++) liberarAST(raiz->filho[i]);
    liberarAST(raiz->irmao);
    free(raiz);
}

    totalErros = 0;

static const char* nomeFase(FaseCompilacao fase) {
    switch (fase) {
        case FASE_LEXICA:    return "LEXICO";
        case FASE_SINTATICA: return "SINTATICO";
        case FASE_SEMANTICA: return "SEMANTICO";
        default:             return "COMPILADOR";
    }
}

void reportarErro(FaseCompilacao fase, int linha, const char *mensagem, const char *detalhe) {
    totalErros++;
    fprintf(stderr, "\n[ERRO %s] Linha %d: %s\n", nomeFase(fase), linha, mensagem);
    if (detalhe != NULL && detalhe[0] != '\0') {
        fprintf(stderr, "   -> Detalhe: %s\n", detalhe);
    }
}

int obterTotalErros(void) {
    return totalErros;
}

void imprimirResumoErros(void) {
    printf("\n============================================================\n");
    printf(" RELATORIO DE COMPILACAO | Total de Erros: %d\n", totalErros);
    printf(" Status: %s\n", totalErros == 0 ? "SUCESSO" : "FALHA NA COMPILACAO");
    printf("============================================================\n");
}

#include "scan.h"

int lineno = 1;
char tokenString[MAXTOKENLEN + 1];

int pegaChar(void)
{
    int c = fgetc(Fonte);

    if (c == '\n')
        lineno++;

    return c;
}

void devolveChar(int c)
{
    if (c == '\n')
        lineno--;

    if (c != EOF)
        ungetc(c, Fonte);
}

TokenType reservado(char *s)
{
    if (!strcmp(s, "else"))   return ELSE;
    if (!strcmp(s, "if"))     return IF;
    if (!strcmp(s, "int"))    return INT;
    if (!strcmp(s, "return")) return RETURN;
    if (!strcmp(s, "void"))   return VOID;
    if (!strcmp(s, "while"))  return WHILE;

    return IDENTIFICADOR;
}

TokenType getToken(void)
{
    int idx;
    int c;
    int p;

    while (1)
    {
        idx = 0;
        c = pegaChar();

        if (c == ' ' || c == '\t' || c == '\n' || c == '\r')
            continue;

        if (c == EOF)
        {
            tokenString[0] = '\0';
            return ENDFILE;
        }

        if (c == '/')
        {
            p = pegaChar();

            if (p == '*')
            {
                while (1)
                {
                    c = pegaChar();

                    if (c == EOF)
                    {
                        tokenString[0] = '\0';
                        return ENDFILE;
                    }

                    if (c == '*')
                    {
                        p = pegaChar();

                        if (p == '/')
                            break;

                        devolveChar(p);
                    }
                }

                continue;
            }

            devolveChar(p);

            tokenString[0] = '/';
            tokenString[1] = '\0';

            return DIVISAO;
        }

        if (isdigit(c))
        {
            while (isdigit(c))
            {
                if (idx < MAXTOKENLEN)
                    tokenString[idx++] = c;

                c = pegaChar();
            }

            devolveChar(c);

            tokenString[idx] = '\0';

            return NUMERO;
        }

        if (isalpha(c))
        {
            while (isalpha(c))
            {
                if (idx < MAXTOKENLEN)
                    tokenString[idx++] = c;

                c = pegaChar();
            }

            devolveChar(c);

            tokenString[idx] = '\0';

            return reservado(tokenString);
        }

        tokenString[0] = c;
        tokenString[1] = '\0';

        switch (c)
        {
            case '=':
                p = pegaChar();

                if (p == '=')
                {
                    tokenString[1] = '=';
                    tokenString[2] = '\0';
                    return IGUAL;
                }

                devolveChar(p);
                return ATRIBUICAO;

            case '<':
                p = pegaChar();

                if (p == '=')
                {
                    tokenString[1] = '=';
                    tokenString[2] = '\0';
                    return MENOR_IGUAL;
                }

                devolveChar(p);
                return MENOR;

            case '>':
                p = pegaChar();

                if (p == '=')
                {
                    tokenString[1] = '=';
                    tokenString[2] = '\0';
                    return MAIOR_IGUAL;
                }

                devolveChar(p);
                return MAIOR;

            case '!':
                p = pegaChar();

                if (p == '=')
                {
                    tokenString[1] = '=';
                    tokenString[2] = '\0';
                    return DIFERENTE;
                }

                devolveChar(p);
                return ERROR;

            case '+':
                return MAIS;

            case '-':
                return MENOS;

            case '*':
                return VEZES;

            case ';':
                return PONTO_VIRGULA;

            case ',':
                return VIRGULA;

            case '(':
                return ABRE_PARENTESE;

            case ')':
                return FECHA_PARENTESE;

            case '[':
                return ABRE_COLCHETE;

            case ']':
                return FECHA_COLCHETE;

            case '{':
                return ABRE_CHAVE;

            case '}':
                return FECHA_CHAVE;

            default:
                return ERROR;
        }
    }
}

==> tabela.c <==
#include "tabela.h"
#include <stdio.h>
#include <string.h>

static Simbolo tabela[MAX_SIMBOLOS];
static int totalSimbolos = 0;

static int pilhaEscopos[MAX_ESCOPOS];
static char nomesEscopos[MAX_ESCOPOS][64];
static int topoEscopo = -1;

void inicializarTabelaSimbolos(void) {
    totalSimbolos = 0;
    topoEscopo = -1;
    entrarEscopo("global");
}

void entrarEscopo(const char *nomeEscopo) {
    if (topoEscopo + 1 >= MAX_ESCOPOS) return;
    topoEscopo++;
    pilhaEscopos[topoEscopo] = totalSimbolos;
    strncpy(nomesEscopos[topoEscopo], nomeEscopo ? nomeEscopo : "bloco", 63);
    nomesEscopos[topoEscopo][63] = '\0';
}

void sairEscopo(void) {
    if (topoEscopo > 0) topoEscopo--;
}

Simbolo* buscarSimboloEscopoAtual(const char *nome) {
    if (topoEscopo < 0) return NULL;
    for (int i = totalSimbolos - 1; i >= pilhaEscopos[topoEscopo]; i--) {
        if (tabela[i].nivelEscopo == topoEscopo && strcmp(tabela[i].nome, nome) == 0) {
            return &tabela[i];
        }
    }
    return NULL;
}

Simbolo* buscarSimbolo(const char *nome) {
    for (int nivel = topoEscopo; nivel >= 0; nivel--) {
        for (int i = totalSimbolos - 1; i >= pilhaEscopos[nivel]; i--) {
            if (tabela[i].nivelEscopo == nivel && strcmp(tabela[i].nome, nome) == 0) {
                return &tabela[i];
            }
        }
    }
    return NULL;
}

bool inserirSimbolo(const char *nome, TipoDado tipo, CategoriaSimbolo categoria, int tamanhoArray, int linha) {
    if (totalSimbolos >= MAX_SIMBOLOS || buscarSimboloEscopoAtual(nome) != NULL) {
        return false;
    }

    Simbolo *s = &tabela[totalSimbolos++];
    strncpy(s->nome, nome, 63);
    s->nome[63] = '\0';
    s->tipo = (categoria == CAT_FUNCAO) ? TipoFuncao : tipo;
    s->tamanhoArray = tamanhoArray;
    s->tipoRetorno = (categoria == CAT_FUNCAO) ? tipo : TipoVoid;
    s->qtdParams = 0;
    s->categoria = categoria;
    s->linhaDeclaracao = linha;
    s->nivelEscopo = topoEscopo;
    strncpy(s->nomeEscopo, nomesEscopos[topoEscopo], 63);
    s->nomeEscopo[63] = '\0';
    return true;
}

void registrarParametroFuncao(Simbolo *funcSimb, TipoDado tipoParam) {
    if (funcSimb != NULL && funcSimb->qtdParams < MAX_PARAMS) {
        funcSimb->tiposParams[funcSimb->qtdParams++] = tipoParam;
    }
}

const char* nomeTipo(TipoDado tipo) {
    switch (tipo) {
        case TipoInt:    return "TipoInt";
        case TipoBool:   return "TipoBool";
        case TipoVoid:   return "TipoVoid";
        case TipoArray:  return "TipoArray(int)";
        case TipoFuncao: return "TipoFuncao";
        case TipoErro:   return "TipoErro";
        default:         return "Desconhecido";
    }
}

void imprimirTabelaSimbolos(void) {
    printf("\n================================================================================\n");
    printf(" TABELA DE SIMBOLOS\n");
    printf("================================================================================\n");
    printf(" %-15s | %-10s | %-15s | %-12s | %-5s | %s\n",
           "Identificador", "Categoria", "Tipo", "Escopo", "Linha", "Detalhes");
    printf("--------------------------------------------------------------------------------\n");

    for (int i = 0; i < totalSimbolos; i++) {
        Simbolo *s = &tabela[i];
        const char *cat = (s->categoria == CAT_FUNCAO) ? "Funcao" :
                          (s->categoria == CAT_PARAMETRO) ? "Parametro" : "Variavel";
        char extra[64] = "-";

        if (s->tipo == TipoArray) {
            snprintf(extra, sizeof(extra), "tam=%d", s->tamanhoArray);
        } else if (s->categoria == CAT_FUNCAO) {
            snprintf(extra, sizeof(extra), "ret=%s, params=%d", nomeTipo(s->tipoRetorno), s->qtdParams);
        }

        printf(" %-15s | %-10s | %-15s | %-12s | %-5d | %s\n",
               s->nome, cat, nomeTipo(s->tipo), s->nomeEscopo, s->linhaDeclaracao, extra);
    }
    printf("================================================================================\n");
}
